cmake_minimum_required(VERSION 3.10)
project(armcontrol)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)

set(ARMCOMM "AUTO" CACHE STRING "Communication backend: AUTO, CAN, SERIAL")
set_property(CACHE ARMCOMM PROPERTY STRINGS AUTO CAN SERIAL)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" SYS_ARCH_LOWER)
if(ARMCOMM STREQUAL "AUTO")
  if(SYS_ARCH_LOWER MATCHES "^(aarch64|arm64|armv7|armv8|arm)$")
    set(ARMCOMM_EFFECTIVE "CAN")
  else()
    set(ARMCOMM_EFFECTIVE "SERIAL")
  endif()
else()
  set(ARMCOMM_EFFECTIVE "${ARMCOMM}")
endif()

add_executable(arm_control_node src/arm_control_node.cpp)
target_include_directories(arm_control_node PRIVATE include)
ament_target_dependencies(arm_control_node rclcpp std_msgs sensor_msgs ament_index_cpp)
target_link_libraries(arm_control_node yaml-cpp Threads::Threads)
if(ARMCOMM_EFFECTIVE STREQUAL "CAN")
  target_sources(arm_control_node PRIVATE src/socketcan.cpp)
  target_compile_definitions(arm_control_node PRIVATE USE_CAN_COMMUNICATION)
elseif(NOT ARMCOMM_EFFECTIVE STREQUAL "SERIAL")
  message(FATAL_ERROR "Invalid ARMCOMM value: ${ARMCOMM}")
endif()
message(STATUS "armcontrol: ${ARMCOMM} -> ${ARMCOMM_EFFECTIVE}")
install(TARGETS arm_control_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_motor_protocol test/test_motor_protocol.cpp)
  target_include_directories(test_motor_protocol PRIVATE include)
  target_link_libraries(test_motor_protocol Threads::Threads util)
  add_test(NAME motor_protocol COMMAND test_motor_protocol)
endif()
ament_package()
